

===== alerts =====
CREATE TABLE alerts(

    id INTEGER PRIMARY KEY AUTOINCREMENT,

    ip TEXT,

    severity TEXT,

    title TEXT,

    description TEXT,

    recommendation TEXT,

    scan_time TEXT

, target TEXT, alert_type TEXT, message TEXT, created_at TEXT, scan_id TEXT, user_id INTEGER DEFAULT 1)
